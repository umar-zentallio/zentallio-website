Cookie banner snippet

Last updated 1 July 2026Zentallio · The AI Decision Platform

Paste the block below just before </body> on every page of the site. It's self-contained (its own styles and script), blocks analytics/marketing until consent, and exposes window.zentallioCookiePrefs() to reopen the settings. The live banner on this page is the same code.

How it works: on first visit the card appears and nothing non-essential loads. “Accept all”, “Essentials only”, or “Customize” (per-category chips, then “Save choices”) all store the choice and fire a zentallio:consent event. Gate your analytics/marketing tags on that event — see the commented hooks inside the script. A “Cookie preferences” link is already wired into the footer of every legal page.

<script>
(function(){
  var KEY='zentallio_cookie_consent';
  var el=document.getElementById('zckBanner');
  if(!el)return;
  var panel=el.querySelector('.zck-panel');
  var acceptBtn=el.querySelector('[data-zck=accept]');
  var mem=null;
  function read(){try{var r=localStorage.getItem(KEY);return r?JSON.parse(r):mem;}catch(e){return mem;}}
  function save(p){mem=p;try{localStorage.setItem(KEY,JSON.stringify(p));}catch(e){}}
  function apply(p){
    // Consent gate — initialise non-essential tags only when allowed:
    // if(p.analytics){ /* load analytics */ }
    // if(p.marketing){ /* load marketing/ad tags */ }
    try{document.dispatchEvent(new CustomEvent('zentallio:consent',{detail:p}));}catch(e){}
  }
  function syncChips(){el.querySelectorAll('.zck-chip input[data-cat]').forEach(function(i){i.closest('.zck-chip').classList.toggle('on',i.checked);});}
  function show(openPanel){
    var cur=read()||{analytics:false,marketing:false};
    el.querySelectorAll('input[data-cat]').forEach(function(i){i.checked=!!cur[i.getAttribute('data-cat')];});
    syncChips();
    panel.hidden=!openPanel;
    el.hidden=false;
  }
  function decide(p){p.essential=true;p.ts=Date.now();save(p);apply(p);el.hidden=true;}
  el.addEventListener('change',function(e){if(e.target.matches('input[data-cat]'))syncChips();});
  el.addEventListener('click',function(e){
    var b=e.target.closest('[data-zck]');if(!b)return;
    var a=b.getAttribute('data-zck');
    if(a==='accept')decide({analytics:true,marketing:true});
    else if(a==='reject')decide({analytics:false,marketing:false});
    else if(a==='manage'){
      var opening=panel.hidden;panel.hidden=!opening;
      // when customizing, the primary action becomes "Save my choices"
      acceptBtn.textContent=opening?'Save choices':'Accept all';
      acceptBtn.setAttribute('data-zck',opening?'save':'accept');
    }
  });
  el.addEventListener('click',function(e){
    var b=e.target.closest('[data-zck=save]');if(!b)return;
    var p={};el.querySelectorAll('input[data-cat]').forEach(function(i){p[i.getAttribute('data-cat')]=i.checked;});
    decide(p);
  });
  window.zentallioCookiePrefs=function(){
    acceptBtn.textContent='Save choices';acceptBtn.setAttribute('data-zck','save');
    show(true);
  };
  var stored=read();
  if(stored){apply(stored);}else{show(false);}
})();
</script>
